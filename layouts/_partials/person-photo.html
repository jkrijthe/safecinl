Without a photo, the person's initials are shown instead. */}} {{- $img := "" }} {{- with .photo }} {{- $img = resources.Get (printf "images/people/%s" .) }} {{- if not $img }}{{ warnf "people.yaml: photo %q not found in assets/images/people/" . }}{{ end }} {{- end }} {{- if $img }} {{- if ne $img.MediaType.SubType "svg" }}{{ $img = $img.Fill "240x240 Top webp" }}{{ end }} {{- else }} {{- $parts := split .name " " }} {{- $initials := substr (index $parts 0) 0 1 }} {{- if gt (len $parts) 1 }}{{ $initials = print $initials (substr (index $parts (sub (len $parts) 1)) 0 1) }}{{ end }} {{ upper $initials }} {{- end }}
